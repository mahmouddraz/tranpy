"""Safe on-disk format for datasets: compressed ``.npz`` arrays plus JSON metadata.

Files are read with ``allow_pickle=False``, so loading a dataset can never execute code.
The JSON header records a format name and version; incompatible files are rejected with
a clear message instead of being half-read.
"""

from __future__ import annotations

import hashlib
import json
import zipfile
from pathlib import Path
from typing import Any

import numpy as np

from tranpy_core.data.dataset import SignalDataset, TabularDataset
from tranpy_core.errors import ChecksumError, DataError

FORMAT_VERSION = 1
_HEADER = "__header__"
_ANNOTATION_PREFIX = "annotation/"

def sha256_file(path: str | Path, chunk_size: int = 1 << 20) -> str:
    """Hex SHA-256 digest of a file."""
    digest = hashlib.sha256()
    with Path(path).open("rb") as fh:
        while chunk := fh.read(chunk_size):
            digest.update(chunk)
    return digest.hexdigest()

def verify_sha256(path: str | Path, expected: str) -> None:
    """Raise :class:`ChecksumError` unless ``path`` has digest ``expected``."""
    actual = sha256_file(path)
    if actual != expected.lower():
        raise ChecksumError(f"checksum mismatch for {path}: expected {expected}, got {actual}")




def _write(path: Path, header: dict[str, Any], arrays: dict[str, np.ndarray]) -> Path:
    """Write an ``.npz`` byte-for-byte reproducibly (fixed member order and timestamps)."""
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_name(path.name + ".tmp")
    members = {_HEADER: np.array(json.dumps(header, sort_keys=True)), **arrays}
    try:
        with zipfile.ZipFile(tmp, "w", compression=zipfile.ZIP_DEFLATED) as zf:
            for name, array in members.items():
                info = zipfile.ZipInfo(name + ".npy", date_time=(1980, 1, 1, 0, 0, 0))
                info.compress_type = zipfile.ZIP_DEFLATED
                with zf.open(info, "w", force_zip64=True) as member:
                    np.lib.format.write_array(
                        member, np.ascontiguousarray(array), allow_pickle=False
                    )
    except BaseException:
        tmp.unlink(missing_ok=True)
        raise
    tmp.replace(path)
    return path


def _read(path: Path, kind: str) -> tuple[dict[str, Any], dict[str, np.ndarray]]:
    try:
        with np.load(path, allow_pickle=False) as npz:
            arrays = {k: npz[k] for k in npz.files}
    except (OSError, ValueError) as exc:
        raise DataError(f"cannot read dataset file {path}: {exc}") from exc
    if _HEADER not in arrays:
        raise DataError(f"{path} is not a tranpy dataset file (no header)")
    header = json.loads(str(arrays.pop(_HEADER).reshape(-1)[0]))
    if header.get("kind") != kind:
        raise DataError(f"{path} holds a {header.get('kind')!r} dataset, expected {kind!r}")
    if header.get("format_version") != FORMAT_VERSION:
        raise DataError(
            f"{path} uses format version {header.get('format_version')}, "
            f"this version of tranpy reads version {FORMAT_VERSION}"
        )
    return header, arrays

def save_tabular(dataset: TabularDataset, path: str | Path) -> Path:
    """Write ``dataset`` to ``path`` (``.npz``)."""
    header = {
        "kind": "tabular",
        "format_version": FORMAT_VERSION,
        "name": dataset.name,
        "description": dataset.description,
        "feature_names": list(dataset.feature_names),
        "label_names": list(dataset.label_names),
        "metadata": dict(dataset.metadata),
    }
    assert dataset.sample_ids is not None
    arrays = {
        "features": dataset.features,
        "labels": dataset.labels,
        "sample_ids": dataset.sample_ids,
    }
    return _write(Path(path), header, arrays)

def load_tabular(path: str | Path) -> TabularDataset:
    """Read a dataset written by :func:`save_tabular`."""

    return TabularDataset(

        labels=arrays["labels"],
        sample_ids=arrays["sample_ids"],
        feature_names=tuple(header["feature_names"]),
        label_names=tuple(header["label_names"]),
        name=header["name"],
        description=header["description"],
        metadata=header["metadata"],

def _encode_decimal(values: np.ndarray, max_decimals: int = 9) -> tuple[np.ndarray, int] | None:
    """Exact scaled-integer encoding of values that are decimals with few digits.

    Returns byte-shuffled first differences (channel-major) of ``round(values * 10**d)``
    for the smallest ``d`` that reproduces every value bit for bit, or None. Decoding
    divides exact integers by an exact power of ten, which is correctly rounded and
    therefore identical to parsing the original decimal text.
    """
    if values.size == 0 or not np.isfinite(values).all():
        return None
    for decimals in range(max_decimals + 1):
        scale = 10.0**decimals
        scaled = np.round(values * scale)
        if np.abs(scaled).max() >= 2.0**53:
            return None
        if np.array_equal(scaled / scale, values):
            ints = np.ascontiguousarray(scaled.astype(np.int64).T)
            delta = np.diff(ints, axis=1, prepend=np.zeros((ints.shape[0], 1), dtype=np.int64))
            shuffled = np.ascontiguousarray(delta.reshape(-1).view(np.uint8).reshape(-1, 8).T)
            return shuffled, decimals
    return None


def _decode_decimal(shuffled: np.ndarray, decimals: int, shape: tuple[int, int]) -> np.ndarray:
    n_rows, n_channels = shape
    delta = np.ascontiguousarray(shuffled.T).view(np.int64).reshape(n_channels, n_rows)
    ints = np.cumsum(delta, axis=1)
    return np.ascontiguousarray((ints / 10.0**decimals).T)

def save_signals(dataset: SignalDataset, path: str | Path, *, compact: bool = True) -> Path:
    """Write ``dataset`` to ``path`` (``.npz``).

    With ``compact`` (default), values that are decimals with at most 9 digits after the
    point (e.g. parsed from CSV exports) are stored as byte-shuffled integer differences,
    typically 2 to 3 times smaller and bit-for-bit identical on reading. Other data are
    stored as plain float64.
    """
    encoded = _encode_decimal(dataset.values) if compact else None
    header = {
        "kind": "signals",
        "format_version": FORMAT_VERSION,
        "name": dataset.name,
        "description": dataset.description,
        "channel_names": list(dataset.channel_names),
        "label_names": list(dataset.label_names),
        "time_unit": dataset.time_unit,
        "metadata": dict(dataset.metadata),
        "values_encoding": (
            {"kind": "float64"}
            if encoded is None
            else {
                "kind": "decimal_delta",
                "decimals": encoded[1],
                "shape": list(dataset.values.shape),
            }
        ),
    }
    assert dataset.sample_ids is not None
    arrays = {
        "time": dataset.time,
        "values": dataset.values if encoded is None else encoded[0],
        "offsets": dataset.offsets,
        "labels": dataset.labels,
        "sample_ids": dataset.sample_ids,
        **{_ANNOTATION_PREFIX + k: v for k, v in dataset.annotations.items()},
    }
    return _write(Path(path), header, arrays)

def load_signals(path: str | Path) -> SignalDataset:
    """Read a dataset written by :func:`save_signals`."""
    header, arrays = _read(Path(path), "signals")
    annotations = {
        k[len(_ANNOTATION_PREFIX) :]: v
        for k, v in arrays.items()
        if k.startswith(_ANNOTATION_PREFIX)
    }
    encoding = header.get("values_encoding", {"kind": "float64"})
    if encoding["kind"] == "decimal_delta":
        values = _decode_decimal(
            arrays["values"], int(encoding["decimals"]), tuple(encoding["shape"])
        )
    elif encoding["kind"] == "float64":
        values = arrays["values"]
    else:
        raise DataError(f"{path}: unknown values encoding {encoding['kind']!r}")
    return SignalDataset(
        time=arrays["time"],
        values=values,
        offsets=arrays["offsets"],
        labels=arrays["labels"],
        sample_ids=arrays["sample_ids"],
        annotations=annotations,
        channel_names=tuple(header["channel_names"]),
        label_names=tuple(header["label_names"]),
        name=header["name"],
        description=header["description"],
        time_unit=header["time_unit"],
        metadata=header["metadata"],
    )

def load_dataset_file(path: str | Path) -> TabularDataset | SignalDataset:
    """Read either kind of dataset file."""
    try:
        return load_tabular(path)
    except DataError as exc:
        if "holds a 'signals' dataset" not in str(exc):
            raise
    return load_signals(path)
